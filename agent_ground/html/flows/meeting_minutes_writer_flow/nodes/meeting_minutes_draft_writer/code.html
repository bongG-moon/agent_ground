48def _extract_text(value: Any) -> str:
49 content = getattr(value, "content", value)

74def _normalize_flattened_table_line(line: str) -> str:
75 if not _TABLE_SEPARATOR_ROW_RE.search(line):
76 return line
77 if not _FLATTENED_ROW_BOUNDARY_RE.search(line):
78 return line
79 expanded = _FLATTENED_ROW_BOUNDARY_RE.sub("|\n|", line)
80 rows = [row.strip() for row in expanded.splitlines() if row.strip()]
81 separator_index = next(
82 (
83 index
84 for index, row in enumerate(rows)
85 if len(_markdown_row_cells(row)) >= 2
86 and all(
87 _TABLE_SEPARATOR_CELL_RE.fullmatch(cell)
88 for cell in _markdown_row_cells(row)
89 )
90 ),
91 -1,
92 )
93 if separator_index != 1:
94 return line
95 separator_cells = _markdown_row_cells(rows[separator_index])
96 column_count = len(separator_cells)
97 header_cells = _markdown_row_cells(rows[0])
98 heading = ""
99 if len(header_cells) == column_count + 1 and not rows[0].lstrip().startswith("|"):
100 heading = header_cells.pop(0).strip()
101 if len(header_cells) != column_count:
102 return line
103 normalized_rows = [header_cells, ["---"] * column_count]
104 for row in rows[separator_index + 1 :]:
105 cells = _markdown_row_cells(row)
106 if len(cells) != column_count:
107 return line
108 normalized_rows.append(cells)
109 if len(normalized_rows) < 3:
110 return line
111 table = "\n".join(
112 "| " + " | ".join(cells) + " |"
113 for cells in normalized_rows
114 )
115 if not heading:
116 return table
117 heading_text = heading.rstrip(":：").strip()
118 if not re.match(r"^#{1,6}\s+", heading_text):
119 if _ACTION_TABLE_HEADING_RE.fullmatch(heading_text):
120 heading_text = f"## {heading_text}"
121 return f"{heading_text}\n\n{table}"

124def normalize_markdown_tables(value: Any) -> str:
125 text = str(value or "").replace("\r\n", "\n").replace("\r", "\n").strip()
126 if not text:
127 return ""
128 result: list[str] = []
129 in_code_fence = False
130 for line in text.splitlines():
131 if line.lstrip().startswith("```"):
132 in_code_fence = not in_code_fence
133 result.append(line.rstrip())
134 continue
135 normalized = line if in_code_fence else _normalize_flattened_table_line(line)
136 if "\n" not in normalized:
137 result.append(normalized.rstrip())
138 continue
139 if result and result[-1].strip():
140 result.append("")
141 result.extend(normalized.splitlines())
142 result.append("")
143 return re.sub(r"\n{3,}", "\n\n", "\n".join(result)).strip()

146def _clean_minutes(value: Any) -> str:
147 text = _extract_text(value).replace("\x00", " ").strip()
148 fenced = re.fullmatch(r"```(?:markdown|md|text)?\s*([\s\S]*?)\s*```", text, re.I)
149 if fenced:
150 text = fenced.group(1).strip()
151 text = normalize_markdown_tables(text)
152 if not text:
153 raise ValueError("회의록 작성 모델이 빈 결과를 반환했습니다.")
154 if len(text) > 200_000:
155 raise ValueError("생성된 회의록이 허용된 최대 길이를 초과했습니다.")
156 return text

1761. 현재 녹취에 실제로 존재하는 사실과 맥락

243
