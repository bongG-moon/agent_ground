44def _extract_text(value: Any) -> str:
45 content = getattr(value, "content", value)
46 if isinstance(content, str):
47 return content
48 if isinstance(content, list):
49 result = []

61def parse_json_object(value: Any) -> dict[str, Any]:
62 text = _extract_text(value).strip()
63 if not text:
64 raise ValueError("스타일 분석 모델 응답이 비어 있습니다.")
65 decoder = json.JSONDecoder()
66 candidates = [re.sub(r"^```(?:json)?\s*|\s*```$", "", text, flags=re.I), text]
67 for candidate in candidates:
68 try:
69 parsed = json.loads(candidate)
70 if isinstance(parsed, dict):
71 return parsed
72 except Exception:
73 pass
74 for index, char in enumerate(candidate):
75 if char != "{":
76 continue
77 try:
78 parsed, _ = decoder.raw_decode(candidate[index:])
79 except Exception:
80 continue
81 if isinstance(parsed, dict):
82 return parsed
83 raise ValueError("스타일 분석 응답에서 JSON object를 찾지 못했습니다.")

86def _safe_text(value: Any, maximum: int) -> str:
87 text = str(value or "").replace("\x00", " ")
88 text = re.sub(r"https?://\S+|[\w.+-]+@[\w.-]+", "[링크]", text, flags=re.I)
89 text = re.sub(r"\b\d{2,}\b", "[값]", text)
90 text = re.sub(r"[\r\t]+", " ", text)
91 return re.sub(r" {2,}", " ", text).strip()[:maximum]

94def _safe_list(value: Any, *, maximum_items: int, maximum_chars: int) -> list[str]:
95 if not isinstance(value, list):
96 return []
97 result: list[str] = []
98 for item in value:
99 text = _safe_text(item, maximum_chars)
100 if text and text not in result:
101 result.append(text)
102 if len(result) >= maximum_items:
103 break
104 return result

122def build_style_prompt(request_value: Any) -> HumanMessage:

149
150중요한 보안·근거 규칙:

180def normalize_style_profile(value: Any, *, pair_count: int) -> dict[str, Any]:
181 parsed = parse_json_object(value)
182 section_order = _safe_list(parsed.get("section_order"), maximum_items=12, maximum_chars=80)
183 selection_rules = _safe_list(parsed.get("selection_rules"), maximum_items=15, maximum_chars=240)
184 omission_rules = _safe_list(parsed.get("omission_rules"), maximum_items=15, maximum_chars=240)
185 phrasing = _safe_list(parsed.get("representative_phrasing"), maximum_items=8, maximum_chars=180)
186 warnings = _safe_list(parsed.get("warnings"), maximum_items=10, maximum_chars=240)
187 if not section_order:
188 section_order = ["회의 개요", "주요 논의", "의사결정", "후속 조치"]
189 warnings.append("섹션 순서가 없어 안전한 기본 순서를 사용했습니다.")
190 if not selection_rules:
191 selection_rules = ["결론, 근거, 의사결정과 실행에 필요한 내용을 우선 기록합니다."]
192 if not omission_rules:
193 omission_rules = ["인사말, 반복, 잡담과 결론에 영향을 주지 않는 발언은 제외합니다."]
194 return {

229 if model is None:

246 ) from exc
247 return {

249 "style_profile": profile,
250 "errors": [],
251 "warnings": deepcopy(profile.get("warnings", [])),
