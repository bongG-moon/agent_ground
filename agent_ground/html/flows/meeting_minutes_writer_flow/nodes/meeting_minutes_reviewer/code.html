115def _normalize_flattened_table_line(line: str) -> str:
116 """한 줄로 붙은 Markdown 표를 행 단위 표로 복구한다."""
117
118 if not _TABLE_SEPARATOR_ROW_RE.search(line):
119 return line
120 if not _FLATTENED_ROW_BOUNDARY_RE.search(line):
121 return line
122
123 expanded = _FLATTENED_ROW_BOUNDARY_RE.sub("|\n|", line)
124 rows = [row.strip() for row in expanded.splitlines() if row.strip()]
125 separator_index = next(
126 (
127 index
128 for index, row in enumerate(rows)
129 if len(_markdown_row_cells(row)) >= 2
130 and all(
131 _TABLE_SEPARATOR_CELL_RE.fullmatch(cell)
132 for cell in _markdown_row_cells(row)
133 )
134 ),
135 -1,
136 )
137 if separator_index != 1:
138 return line
139
140 separator_cells = _markdown_row_cells(rows[separator_index])
141 column_count = len(separator_cells)
142 header_cells = _markdown_row_cells(rows[0])
143 heading = ""
144 if len(header_cells) == column_count + 1 and not rows[0].lstrip().startswith("|"):
145 heading = header_cells.pop(0).strip()
146 if len(header_cells) != column_count:
147 return line
148
149 normalized_rows = [header_cells, ["---"] * column_count]
150 for row in rows[separator_index + 1 :]:
151 cells = _markdown_row_cells(row)
152 if len(cells) != column_count:
153 return line
154 normalized_rows.append(cells)
155 if len(normalized_rows) < 3:
156 return line
157
158 table = "\n".join(
159 "| " + " | ".join(cells) + " |"
160 for cells in normalized_rows
161 )
162 if not heading:
163 return table
164 heading_text = heading.rstrip(":：").strip()
165 if not re.match(r"^#{1,6}\s+", heading_text):
166 if _ACTION_TABLE_HEADING_RE.fullmatch(heading_text):
167 heading_text = f"## {heading_text}"
168 return f"{heading_text}\n\n{table}"
